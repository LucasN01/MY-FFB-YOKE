FILE* logFile=nullptr;
void Log(const char* format, ...) {
    va_list args; va_start(args,format); std::vprintf(format,args); va_end(args); std::fflush(stdout);
    if(logFile) { va_start(args,format); std::vfprintf(logFile,format,args); va_end(args); std::fflush(logFile); }
}
struct Telemetry { double pressurePSF, ias, windX, windY, windZ, ground, slew;
    double aoa, stallAlpha, flapLeft, flapRight, vs0, vs1, gear, retractable, spoilerLeft, spoilerRight, groundSpeed, yokeRoll, yokePitch; };
HANDLE sim=nullptr, port=INVALID_HANDLE_VALUE;
std::atomic<bool> quit{false};
bool failed=false, paused=true, running=false, opened=false, enable=false, gusts=false, buffets=true;
bool pauseKnown=false; // Startup default is not confirmation of simulator pause.
unsigned samples=0;
uint16_t sequence=0;
ULONGLONG lastData=0, lastSend=0;
double referenceKnots=120, gustGain=1, gustLongitudinal=0.25;
FILE* csv=nullptr;
Telemetry latest{};
YokeGust::Model gustModel;
BOOL WINAPI Stop(DWORD) { quit=true; return TRUE; }
bool Check(HRESULT hr, const char* op) {
    if(SUCCEEDED(hr)) return true;
    Log("ERROR %s 0x%08lX\n",op,(unsigned long)hr); failed=quit=true; return false;
}
void Send(bool armed, double q, double r=0, double p=0,
          YokeBuffet::Output buffet=YokeBuffet::Output{}, double groundEnvelope=0, double yokeRoll=0, double yokePitch=0) {
    if(port==INVALID_HANDLE_VALUE) return;
    const bool confirmedPause=opened && pauseKnown && paused && !failed && !quit;
    uint8_t frame[26]={0xa5,0x5a,5,uint8_t(confirmedPause ? 2 : (armed?1:0))};
    YokeSC::put16(frame+4,++sequence);
    YokeSC::put16(frame+6,uint16_t(YokeSC::clip(float(q),0,1)*10000));
    YokeSC::put16(frame+8,uint16_t(int16_t(YokeSC::clip(float(r),-1,1)*10000)));
    YokeSC::put16(frame+10,uint16_t(int16_t(YokeSC::clip(float(p),-1,1)*10000)));
    YokeSC::put16(frame+12,uint16_t(buffet.stall*10000));
    YokeSC::put16(frame+14,uint16_t(buffet.gear*10000));
    YokeSC::put16(frame+16,uint16_t(buffet.spoiler*10000));
    YokeSC::put16(frame+18,uint16_t(groundEnvelope*10000));
    YokeSC::put16(frame+20,uint16_t(int16_t(yokeRoll*10000)));
    YokeSC::put16(frame+22,uint16_t(int16_t(yokePitch*10000)));
    YokeSC::put16(frame+24,YokeSC::crc(frame,24));
    DWORD written=0;
    if(!WriteFile(port,frame,sizeof(frame),&written,nullptr)||written!=sizeof(frame)) {
        Log("ERROR serial write; watchdog del firmware retirara suplementos.\n"); failed=quit=true;
    }
    lastSend=GetTickCount64();
}
void CALLBACK Receive(SIMCONNECT_RECV* data,DWORD bytes,void*) {
    switch(data->dwID) {
    case SIMCONNECT_RECV_ID_OPEN: {
        opened=true;
        auto* o=reinterpret_cast<SIMCONNECT_RECV_OPEN*>(data);
        Log("Conectado: %s %lu.%lu\n",o->szApplicationName,o->dwApplicationVersionMajor,o->dwApplicationVersionMinor);
        const char* names[]={"DYNAMIC PRESSURE","AIRSPEED INDICATED","AIRCRAFT WIND X","AIRCRAFT WIND Y","AIRCRAFT WIND Z","SIM ON GROUND","IS SLEW ACTIVE",
            "INCIDENCE ALPHA","STALL ALPHA","TRAILING EDGE FLAPS LEFT PERCENT",
            "TRAILING EDGE FLAPS RIGHT PERCENT","DESIGN SPEED VS0","DESIGN SPEED VS1",
            "GEAR LEFT POSITION","IS GEAR RETRACTABLE","SPOILERS LEFT POSITION","SPOILERS RIGHT POSITION","GROUND VELOCITY","YOKE X POSITION","YOKE Y POSITION"};
        const char* units[]={"psf","knots","knots","knots","knots","Bool","Bool",
            "degrees","degrees","percent over 100","percent over 100","knots","knots",
            "percent over 100","Bool","percent over 100","percent over 100","knots","Position","Position"};
        for(int i=0;i<20;++i) if(!Check(SimConnect_AddToDataDefinition(sim,1,names[i],units[i],SIMCONNECT_DATATYPE_FLOAT64),names[i])) return;
        Check(SimConnect_SubscribeToSystemEvent(sim,2,"Pause"),"Pause");
        Check(SimConnect_SubscribeToSystemEvent(sim,3,"Sim"),"Sim");
        Check(SimConnect_RequestDataOnSimObject(sim,1,1,SIMCONNECT_OBJECT_ID_USER,SIMCONNECT_PERIOD_SIM_FRAME),"telemetry");
        break;
    }
    case SIMCONNECT_RECV_ID_EVENT: {
        auto* e=reinterpret_cast<SIMCONNECT_RECV_EVENT*>(data);
        if(e->uEventID==2) { paused=e->dwData!=0; pauseKnown=true; }
        if(e->uEventID==3) running=e->dwData!=0;
        if(e->uEventID==2 || paused || !running) { Send(false,0); gustModel.reset(); }
        break;
    }
    case SIMCONNECT_RECV_ID_SYSTEM_STATE: {
        auto* e=reinterpret_cast<SIMCONNECT_RECV_SYSTEM_STATE*>(data);
        if(e->dwRequestID==2) { paused=e->dwInteger!=0; pauseKnown=true; }
        if(e->dwRequestID==3) running=e->dwInteger!=0;
        break;
    }
    case SIMCONNECT_RECV_ID_SIMOBJECT_DATA: {
        auto* d=reinterpret_cast<SIMCONNECT_RECV_SIMOBJECT_DATA*>(data);
        const size_t offset=reinterpret_cast<char*>(&d->dwData)-reinterpret_cast<char*>(data);
        if(d->dwRequestID!=1||d->dwDefineCount!=20||bytes<offset+sizeof(Telemetry)) return;
        Telemetry t{}; std::memcpy(&t,&d->dwData,sizeof(t));
        if(!std::isfinite(t.pressurePSF)||t.pressurePSF<0||t.pressurePSF>5000||
           !std::isfinite(t.yokeRoll)||t.yokeRoll < -1||t.yokeRoll > 1||!std::isfinite(t.yokePitch)||t.yokePitch < -1||t.yokePitch > 1||
           !std::isfinite(t.ias)||!std::isfinite(t.windX)||!std::isfinite(t.windY)||!std::isfinite(t.windZ)||
           !std::isfinite(t.ground)||(t.ground!=0 && t.ground!=1)||!std::isfinite(t.slew)) { Send(false,0); gustModel.reset(); return; }
        const ULONGLONG now=GetTickCount64();
        const double dt=lastData?(now-lastData)*.001:0;
        const bool fresh=lastData && now-lastData<250;
        lastData=now; latest=t; ++samples;
        const double qPa=t.pressurePSF*47.88025898;
        const double qRef=.5*1.225*std::pow(referenceKnots*.514444444,2);
        const double q=YokeSC::clip(float(qPa/qRef),0,1);
        const bool validState=running&&!paused&&t.slew==0;
        const bool active=enable&&validState&&fresh;
        // Monitor mode evaluates the same model without arming serial output.
        const bool modelActive=gusts&&validState&&fresh&&t.ground==0;
        const YokeGust::Force force=gustModel.update(t.windX,t.windY,t.windZ,
            dt,q,gustGain,gustLongitudinal,modelActive);
        const double r=force.roll, p=force.pitch;
        const YokeBuffet::Input buffetInput={t.ias,t.aoa,t.stallAlpha,t.flapLeft,
            t.flapRight,t.vs0,t.vs1,t.gear,t.retractable,t.spoilerLeft,t.spoilerRight};
        const auto buffet=YokeBuffet::calculate(buffetInput,
            buffets&&validState&&fresh&&t.ground==0);
        const double groundEnvelope=YokeGround::envelope(t.ground,t.groundSpeed,validState&&fresh);
        if(now-lastSend>=20) Send(active,q,r,p,buffet,groundEnvelope,t.yokeRoll,t.yokePitch);
        // Bridge-side indicators, not Arduino/PWM acknowledgements.
        // Independent columns allow simultaneous effects. No armed output => all zero.
        const int turbulenceOn=active && (std::fabs(r)>0.1 || std::fabs(p)>0.1);
        const int stallOn=active && uint16_t(buffet.stall*10000)>0;
        const int gearOn=active && uint16_t(buffet.gear*10000)>0;
        const int spoilerOn=active && uint16_t(buffet.spoiler*10000)>0;
        if(csv) { std::fprintf(csv,"%llu,%.5f,%.2f,%.2f,%.3f,%.3f,%.0f,%d,%d,%d,%.5f,%.5f,%.5f,%.2f,%.3f,%.3f,%.3f,%.3f,%.2f,%.2f,%.3f,%.3f,%.2f,%.2f,%.3f,%.0f,%.3f,%.3f,%d,%d,%d,%d,%.3f,%.5f,%d,%d,%.5f,%.5f\n",now,q,qPa,t.ias,r,p,t.ground,paused,running,active,t.windX,t.windY,t.windZ,gustGain,gustLongitudinal,buffet.stall,buffet.gear,buffet.spoiler,t.aoa,t.stallAlpha,t.flapLeft,t.flapRight,t.vs0,t.vs1,t.gear,t.retractable,t.spoilerLeft,t.spoilerRight,turbulenceOn,stallOn,gearOn,spoilerOn,t.groundSpeed,groundEnvelope,active&&uint16_t(groundEnvelope*10000)>0,active,t.yokeRoll,t.yokePitch); std::fflush(csv); }
        if(samples%60==1) Log("q=%.0f Pa ratio=%.3f IAS=%.1f kt ground=%.0f pause=%d running=%d armed=%d gustR=%+.3f gustP=%+.3f gain=%.1f yokeR=%+.3f yokeP=%+.3f\n",qPa,q,t.ias,t.ground,paused,running,active,r,p,gustGain,t.yokeRoll,t.yokePitch);
        break;
    }
    case SIMCONNECT_RECV_ID_EXCEPTION: {
        auto* e=reinterpret_cast<SIMCONNECT_RECV_EXCEPTION*>(data);
        Log("EXCEPTION=%lu SendID=%lu Index=%lu\n",e->dwException,e->dwSendID,e->dwIndex);
        failed=quit=true; Send(false,0); break;
    }
    case SIMCONNECT_RECV_ID_QUIT: quit=true; Send(false,0); break;
    }
}
bool OpenPort(const std::string& name) {
    if(name.size()<4||name.substr(0,3)!="COM"||name.find_first_not_of("0123456789",3)!=std::string::npos) return false;
    port=CreateFileA(("\\\\.\\"+name).c_str(),GENERIC_READ|GENERIC_WRITE,0,nullptr,OPEN_EXISTING,0,nullptr);
    if(port==INVALID_HANDLE_VALUE) return false;
    DCB dcb{}; dcb.DCBlength=sizeof(dcb);
    if(!GetCommState(port,&dcb)) return false;
    dcb.BaudRate=115200; dcb.ByteSize=8; dcb.Parity=NOPARITY; dcb.StopBits=ONESTOPBIT;
    dcb.fBinary=TRUE; dcb.fParity=FALSE; dcb.fDtrControl=DTR_CONTROL_ENABLE;
    dcb.fOutxCtsFlow=FALSE; dcb.fOutxDsrFlow=FALSE; dcb.fOutX=FALSE; dcb.fInX=FALSE;
    if(!SetCommState(port,&dcb)) return false;
    COMMTIMEOUTS timeouts{}; timeouts.ReadIntervalTimeout=MAXDWORD; timeouts.WriteTotalTimeoutConstant=10;
    if(!SetCommTimeouts(port,&timeouts)) return false;
    PurgeComm(port,PURGE_RXCLEAR|PURGE_TXCLEAR);
    std::string response;
    const ULONGLONG start=GetTickCount64(); ULONGLONG query=0;
    while(GetTickCount64()-start<3000) {
        if(GetTickCount64()-query>200) { DWORD n=0; if(!WriteFile(port,"?",1,&n,nullptr)) return false; query=GetTickCount64(); }
        char b[256]; DWORD n=0; if(!ReadFile(port,b,sizeof(b),&n,nullptr)) return false;
        response.append(b,n);
        if(response.find("YOKE17SC5\n")!=std::string::npos) return true;
        if(response.size()>1024) response.erase(0,response.size()-128);
        Sleep(10);
    }
    return false;
}
int main(int argc,char** argv) {
    std::string com; double duration=20; bool durationSpecified=false; // Double click: 20-second monitor, no motor commands.
    for(int i=1;i<argc;++i) {
        const std::string arg=argv[i];
        if(arg=="--enable") enable=true;
        else if(arg=="--gusts") gusts=true;
        else if(arg=="--no-buffets") buffets=false;
        else if(arg=="--gust-gain"&&i+1<argc) {
            const char* start=argv[++i]; char* end=nullptr;
            gustGain=std::strtod(start,&end);
            if(end==start||*end||!std::isfinite(gustGain)||gustGain<0||gustGain>10) {
                Log("--gust-gain debe ser un numero entre 0 y 10.\n"); return 2;
            }
        }
        else if(arg=="--gust-longitudinal"&&i+1<argc) {
            const char* start=argv[++i]; char* end=nullptr;
            gustLongitudinal=std::strtod(start,&end);
            if(end==start||*end||!std::isfinite(gustLongitudinal)||std::fabs(gustLongitudinal)>1) {
                Log("--gust-longitudinal debe estar entre -1 y 1.\n"); return 2;
            }
        }
        else if(arg=="--port"&&i+1<argc) com=argv[++i];
        else if(arg=="--seconds"&&i+1<argc) { duration=std::atof(argv[++i]); durationSpecified=true; }
        else if(arg=="--reference-knots"&&i+1<argc) referenceKnots=std::atof(argv[++i]);
        else { Log("Uso: Bridge.exe [--seconds 20] [--port COMn --enable] [--gusts --gust-gain 0..10 --gust-longitudinal -1..1] [--reference-knots 120] [--no-buffets]\n"); return 2; }
    }
    if(!com.empty()&&!durationSpecified) duration=0;
    if(!std::isfinite(referenceKnots)||referenceKnots<40||referenceKnots>250||!std::isfinite(duration)||duration<0|| (enable&&com.empty())) return 2;
    char file[MAX_PATH]; GetModuleFileNameA(nullptr,file,MAX_PATH); char* tail=std::strrchr(file,'\\');
    if(tail) { strcpy_s(tail+1,MAX_PATH-(tail+1-file),"bridge.log"); fopen_s(&logFile,file,"w"); }
    if(tail) { strcpy_s(tail+1,MAX_PATH-(tail+1-file),"telemetry.csv"); fopen_s(&csv,file,"w"); }
    if(csv) std::fprintf(csv,"time_ms,q_ratio,q_pa,ias_knots,gust_roll,gust_pitch,on_ground,paused,running,armed,wind_x_knots,wind_y_knots,wind_z_knots,gust_gain,gust_longitudinal,stall_amp,gear_amp,spoiler_amp,aoa_deg,stall_alpha_deg,flap_left,flap_right,vs0_knots,vs1_knots,gear_fraction,gear_retractable,spoiler_left,spoiler_right,Turbulencia,Buffet_Perdida,Buffet_Tren,Buffet_Spoilers,ground_speed_knots,ground_amp,Vibracion_Tierra,Centrado_Disponible,yoke_roll,yoke_pitch\n");
    Log("Yoke efectos SimConnect SC5 | FSX SP2 Win32 | %s | qRef=%.1f kt (ajuste, NO Vne)\n",enable?"SUPLEMENTOS ACTIVADOS":"SOLO MONITOR",referenceKnots);
    Log("Viento: %s ganancia=%.2f (limite de senal +/-1; depende del firmware cargado)\n",gusts?"ON":"OFF",gustGain);
    Log("Modelo: tauHP=1s tauLP=0.08s escala=5kt roll=+0.70 longitudinal=%+.3f\n",gustLongitudinal);
    Log("Buffet perdida/tren/spoilers: %s (banderas finales en Arduino)\n",buffets?"ON":"OFF");
    if(!com.empty()&&!OpenPort(com)) { Log("No se identifico firmware YOKE17SC5 en %s. No se enviaron fuerzas.\n",com.c_str()); if(port!=INVALID_HANDLE_VALUE)CloseHandle(port); if(csv)std::fclose(csv); return 3; }
    SetConsoleCtrlHandler(Stop,TRUE);
    const ULONGLONG start=GetTickCount64();
    if(Check(SimConnect_Open(&sim,"Yoke 1.7 gust bridge",nullptr,0,nullptr,SIMCONNECT_OPEN_CONFIGINDEX_LOCAL),"Open")) {
        while(!quit) {
            if(!Check(SimConnect_CallDispatch(sim,Receive,nullptr),"Dispatch")) break;
            const ULONGLONG now=GetTickCount64();
            if((!lastData||now-lastData>250||paused||!running)&&now-lastSend>100) { Send(false,0); gustModel.reset(); }
            if(!opened&&now-start>15000) { failed=true; break; }
            if(duration>0&&now-start>=duration*1000) break;
            if(port!=INVALID_HANDLE_VALUE) { char b[256]; DWORD n; if(!ReadFile(port,b,sizeof(b),&n,nullptr)){ failed=true; break; } }
            Sleep(5);
        }
        opened=false; Send(false,0); SimConnect_Close(sim);
    }
    if(port!=INVALID_HANDLE_VALUE) CloseHandle(port);
    if(csv) std::fclose(csv);
    Log("Fin: muestras=%u error=%d\n",samples,failed);
    return failed||samples==0?1:0;
}
